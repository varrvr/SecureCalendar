cmake_minimum_required(VERSION 3.22)
project(secure_calendar_mapping LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(BUILD_CMD OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_FAST OFF CACHE BOOL "" FORCE)
if(NOT BEE2_SOURCE_DIR)
    message(FATAL_ERROR "Set BEE2_SOURCE_DIR to the pinned Bee2 checkout")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${BEE2_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE BEE2_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT BEE2_REV STREQUAL "4341f7a5c300f8e6dc292559e671eb899d831433")
    message(FATAL_ERROR "Unexpected Bee2 revision; do not silently change the mapping implementation")
endif()
add_subdirectory("${BEE2_SOURCE_DIR}" bee2 EXCLUDE_FROM_ALL)
if(ANDROID)
    # Bionic provides pthread symbols in libc; Android NDK has no standalone libpthread.
    foreach(property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
        get_target_property(libraries bee2_static ${property})
        if(libraries)
            list(REMOVE_ITEM libraries pthread)
            set_property(TARGET bee2_static PROPERTY ${property} "${libraries}")
        endif()
    endforeach()
endif()
add_library(sc_mapping_core STATIC mapping.c oprf_point.c hash_group.c)
target_include_directories(sc_mapping_core PRIVATE "${BEE2_SOURCE_DIR}/include" "${BEE2_SOURCE_DIR}/src/crypto/bign")
target_link_libraries(sc_mapping_core PRIVATE bee2_static)
target_compile_options(sc_mapping_core PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
    add_library(bee2mapping SHARED mapping_jni.c)
    target_link_libraries(bee2mapping PRIVATE sc_mapping_core)
    target_include_directories(bee2mapping PRIVATE "${BEE2_SOURCE_DIR}/include" "${BEE2_SOURCE_DIR}/src/crypto/bign")
    target_link_options(bee2mapping PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")
else()
    enable_testing()
    add_library(sc_hash_reference SHARED hash_group.c)
    target_include_directories(sc_hash_reference PRIVATE "${BEE2_SOURCE_DIR}/include" "${BEE2_SOURCE_DIR}/src/crypto/bign")
    target_link_libraries(sc_hash_reference PRIVATE bee2_static)
    set_target_properties(sc_hash_reference PROPERTIES C_VISIBILITY_PRESET default)
    add_executable(mapping_vectors mapping_vectors.c)
    target_link_libraries(mapping_vectors PRIVATE sc_mapping_core)
    add_test(NAME mapping_standard_vector_and_invalid_points COMMAND mapping_vectors)
    add_executable(oprf_point_vectors oprf_point_vectors.c)
    target_link_libraries(oprf_point_vectors PRIVATE sc_mapping_core)
    add_test(NAME oprf_group_blind_evaluate_unblind COMMAND oprf_point_vectors)
endif()
